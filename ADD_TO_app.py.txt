import os
from flask import render_template

POKEMON_DB = os.path.join(os.path.dirname(os.path.abspath(__file__)), "pokemon_quest.duckdb")
_pokemon_cache = None


@app.route("/pokemon-quest")
def pokemon_quest():
    global _pokemon_cache
    if _pokemon_cache is None:
        import duckdb  # imported here so the rest of the app still runs if duckdb is missing
        with duckdb.connect(POKEMON_DB, read_only=True) as con:
            cur = con.execute(
                "SELECT id, name, type1, type2, clue1, clue2, clue3, answers, image_url "
                "FROM pokemon ORDER BY id"
            )
            cols = [c[0] for c in cur.description]
            rows = [dict(zip(cols, r)) for r in cur.fetchall()]
        for r in rows:
            r["answers"] = r["answers"].split(";")
        _pokemon_cache = rows
    return render_template("pokemon_quest.html", pokemon=_pokemon_cache)
